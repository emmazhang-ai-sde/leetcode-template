使用 BFS 的几个问题

BFS 初始的种子是什么？

队列中存储的结构是什么？

要不要记录层级信息(size)，初始的 level 是 0 还是 1?

是否需要 visited 记录访问过的点，以防止回头？

层级遍历重点

层级遍历时，step+1 需要放在层级遍历结束之后，根据题意 step 的初始值 0 or -1

Level order 遍历时如果要记录 level order 结果的话，要注意最后一轮可能 temp 为空，加入 res 之前要做判定 temp

队列元素 Visited

在入队列的时候更新 visited：起点需要先放入 visited 或者结果中

在出队列的时候更新 visited：入队之前除了要检查 visited，还要 check queue 里有没有重复元素。

最小生成树问题

使用 Heap 做 BFS

  • 入 Heap 之前检测 visited 防止死循环
  • pop 之后如果发现已经访问过或者代价比记录高就 skip
  • 出队列时更新 visited 结果
重点

能够用 BFS 解决的问题，一定不要用 DFS 去做！

def bfs(self, grid, queue, visited):
    step = -1
    while len(queue) > 0:
        size = len(queue)
        for _ in range(size):
            x,y = queue.popleft()
            for dx,dy in [(0,1),(0,-1),(1,0),(-1,0)]:
                newx = x + dx
                newy = y + dy
                if self.isValid(newx,newy, grid, visited):
                    visited.add((newx,newy))
                    queue.append((newx,newy))
        step +=1
    return step

def isValid(self,x,y,grid, visited):
    return 0<=x<len(grid) and 0<=y<len(grid[0]) and (x,y) not in visited and grid[x][y] == 1

def dijkstra(self, heap, n, graph):
    visited = {}

    while len(heap) > 0:
        curr, node = heapq.heappop(heap)

        if node in visited and visited[node] < curr:
            continue
        visited[node] = curr
        if len(visited) == n:
            return curr
        for val, nextNode in graph[node]:
            if nextNode not in visited:
                heapq.heappush(heap, (val + curr, nextNode))

    return -1

public int bfs(Queue<Pair> queue, int[][] grid, boolean[][] visited) {
    int[] rowMovement = {-1, 0, 1, 0};
    int[] colMovement = {0, 1, 0, -1};
    int step = -1;
    while (queue.size() > 0) {
        int size = queue.size();
        for (int count =0; count < size; count++) {
            Pair curr = queue.poll();
            for (int i=0;i<4;i++) {
                int newX = curr.x + rowMovement[i];
                int newY = curr.y + colMovement[i];
                if (this.isValid(newX,newY,grid, visited)) {
                    Pair newPair = new Pair(newX, newY);
                    visited[newX][newY] = true;
                    this.rottedCount++;
                    queue.offer(newPair);
                }
            }
        }
        step+=1;
    }
    return step;
}

public boolean isValid(int x, int y, int[][] grid, boolean[][] visited) {
    return x >=0 && x < grid.length &&
           y >=0 && y < grid[0].length && visited[x][y] == false && grid[x][y] == 1;
}

public void bfs(TreeNode root, List<List<Integer>> res) {
    Queue<TreeNode> queue = new LinkedList<>();
    queue.offer(root);

    while (queue.size() > 0) {
        int size = queue.size();
        List<Integer> temp = new ArrayList<>();
        for (int count =0; count < size; count++) {
            TreeNode curr = queue.poll();
            if (curr != null) {
                temp.add(curr.val);
                queue.offer(curr.left);
                queue.offer(curr.right);
            }
        }
        if (!temp.isEmpty()) {
            res.add(temp);
        }
    }
}

public int dijkstar(Map<Integer, List<MyPair>> graph, int n, int k)
{
    Queue<MyPair> heap = new PriorityQueue<>((pair1, pair2) -> pair1.val - pair2.val);
    heap.add(new MyPair(0,k));
    Map<Integer, Integer> visited = new HashMap<>();
    while (heap.size() > 0) {
        MyPair currPair = heap.poll();
        if (visited.containsKey(currPair.node) && visited.get(currPair.node) < currPair.val) {
            continue;
        }
        visited.put(currPair.node, currPair.val);
        if (visited.size() == n) {
            return currPair.val;
        }
        for (MyPair pair : graph.getOrDefault(currPair.node, new ArrayList<>())) {
            if (!visited.containsKey(pair.node)) {
                heap.add(new MyPair(pair.val + currPair.val, pair.node));
            }
        }
    }
    return -1;
}
