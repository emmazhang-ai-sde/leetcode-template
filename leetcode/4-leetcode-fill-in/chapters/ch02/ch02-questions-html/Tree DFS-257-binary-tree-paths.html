257.Binary Tree Paths

1# Definition for a binary tree node.
2# class TreeNode:
3# def __init__(self, val=0, left=None, right=None):
4# self.val = val
5# self.left = left
6# self.right = right
7
8class Solution:
9 def dfs(self, root) -> List[str]:
10 returnPath = []
11 if not root:
12 return returnPath
13
14 leftReturn = self.dfs(root.left)
15 rightReturn = self.dfs(root.right)
16
17 if root.left is None and root.right is None:
18 return [str(root.val)]
19
20 for list_str in leftReturn:
21 returnPath.append(str(root.val)+ "->" + list_str)
22
23 for list_str in rightReturn:
24 returnPath.append(str(root.val)+ "->" + list_str)
25
26 return returnPath
27
28 def binaryTreePaths(self, root: Optional[TreeNode]) -> List[str]:
29 res = self.dfs(root)
30 return res
