def dfs(self, root):
    if not root:
        return 0

    leftReturn = self.dfs(root.left)
    rightReturn = self.dfs(root.right)

    # Optional Leaf processing
    if root.left is None and root.right is None:
        return 1

    height = max(leftReturn, rightReturn) + 1
    return height

public int dfs(TreeNode root) {
    if (root == null) {
        return 0;
    }

    int leftReturn = dfs(root.left);
    int rightReturn = dfs(root.right);

    // Optional Leaf processing
    if (root.left == null && root.right == null) {
        return 1;
    }
    int height = Math.max(leftReturn, rightReturn) + 1;
    return height;
}
