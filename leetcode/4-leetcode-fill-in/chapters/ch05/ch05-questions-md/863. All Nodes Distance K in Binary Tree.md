# 863. All Nodes Distance K

```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, x):
#         self.val = x
#         self.left = None
#         self.right = None

class Solution:
    def distanceK(self, root: TreeNode, target: TreeNode, k: int) -> List[int]:
        graph = collections.defaultdict(list)
        self.dfs(root, graph, None)

        queue = collections.deque([target])
        visited = set()
        visited.add(target)

        while len(queue) > 0 and k >=0:
            size = len(queue)
            res = []
            for _ in range(size):
                curr = queue.popleft()
                res.append(curr.val)
                for neighbour in graph[curr]:
                    if neighbour not in visited:
                        visited.add(neighbour)
                        queue.append(neighbour)
            k-=1

        if k == -1:
            return res
        else:
            return []

    def dfs(self, node, graph, parent):
        if parent:
            graph[node].append(parent)
        if node.left:
            graph[node].append(node.left)
            self.dfs(node.left, graph, node)
        if node.right:
            graph[node].append(node.right)
            self.dfs(node.right, graph, node)
```


## Input
```python
root = [3, 5, 1, 6, 2, 0, 8, null, null, 7, 4]
target = Node(5)
k = 2
```

## Phase 1 — Build Undirected Graph (DFS)
Goal: Convert the tree into a bidirectional adjacency map so we can traverse upward (child → parent).
```
Original Tree                    graph (after DFS)

        3                        3  → [5, 1]
       / \                       5  → [3, 6, 2]
      5   1                      1  → [3, 0, 8]
     / \ / \                     6  → [5]
    6  2 0  8                    2  → [5, 7, 4]
      / \                        0  → [1]
     7   4                       8  → [1]
                                 7  → [2]
                                 4  → [2]
```                                
Key: Every parent-child edge becomes two entries:
```python
graph[node].append(child)    # downward
graph[node].append(parent)   # upward  ← this is the new part vs. normal tree BFS
```

## Phase 2 — BFS from Target (Layer by Layer)

Goal: Start from target, expand outward one layer per round, stop when k reaches 0.
```
Round 0  (distance = 0)        queue = [Node(5)]          k = 2
         ┌─────────────────────────────────────────────┐
         │ curr = Node(5)                              │
         │ result = [5]                                │
         │ neighbors of 5 → [3, 6, 2]  all unvisited   │
         │ add to queue                                │
         └─────────────────────────────────────────────┘
         queue = [Node(3), Node(6), Node(2)]
         visited = {5, 3, 6, 2}
         k → 1

Round 1  (distance = 1)        queue = [Node(3), Node(6), Node(2)]    k = 1
         ┌─────────────────────────────────────────────┐
         │ curr = Node(3)                              │
         │ result = [3]                                │
         │ neighbors of 3 → [5, 1]                     │
         │   Node(5) already visited → skip            │
         │   Node(1) unvisited → add to queue          │
         ├─────────────────────────────────────────────┤
         │ curr = Node(6)                              │
         │ result = [3, 6]                             │
         │ neighbors of 6 → [5]                        │
         │   Node(5) already visited → skip            │
         ├─────────────────────────────────────────────┤
         │ curr = Node(2)                              │
         │ result = [3, 6, 2]                          │
         │ neighbors of 2 → [5, 7, 4]                  │
         │   Node(5) already visited → skip            │
         │   Node(7), Node(4) unvisited → add to queue │
         └─────────────────────────────────────────────┘
         queue = [Node(1), Node(7), Node(4)]
         visited = {5, 3, 6, 2, 1, 7, 4}
         k → 0

Round 2  (distance = 2)        queue = [Node(1), Node(7), Node(4)]    k = 0
         ┌─────────────────────────────────────────────┐
         │ curr = Node(1), Node(7), Node(4)             │
         │ result = [1, 7, 4]                           │
         │ all neighbors already visited → nothing added│
         └─────────────────────────────────────────────┘
         queue = []
         k → -1   ← loop exits here
```

## Phase 3 — Return Result
```
k == -1  →  return result = [1, 7, 4]  ✓
```
