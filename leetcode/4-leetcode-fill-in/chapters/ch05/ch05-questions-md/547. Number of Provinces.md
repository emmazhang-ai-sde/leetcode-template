# bfs

class Solution:
    def findCircleNum(self, isConnected: List[List[int]]) -> int:
        visited = set()
        ans = 0
        for i in range(len(isConnected)):
            if i in visited:
                continue
            ans += 1
            self.bfs(i, visited, isConnected) 
        return ans

    def bfs(self, i, visited, isConnected):
        import collections
        queue = collections.deque([i])
        visited.add(i)
        while queue:
            curr = queue.popleft()
            for neighbor in range(len(isConnected)):
                if isConnected[curr][neighbor] and neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
```

---

**Key difference from Clone Graph:** In Clone Graph, neighbors are stored explicitly as a list (`node.neighbors`). Here, neighbors are **implicit** — you discover them by scanning the entire row `isConnected[curr][:]` and checking if the value is `1`. The BFS logic is identical; only the neighbor-discovery mechanism differs.
