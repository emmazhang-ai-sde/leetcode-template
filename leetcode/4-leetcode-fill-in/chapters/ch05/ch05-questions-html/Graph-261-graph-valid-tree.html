8 # Create Graph
9 graphs = collections.defaultdict(list)
10 for edge in edges:
11 graphs[edge[0]].append(edge[1])
12 graphs[edge[1]].append(edge[0])
13

21 for neighbour in neighbours:

24 queue.append(neighbour)
25
26 return len(visited) == n
