133.Clone Graph

1"""
2# Definition for a Node.
3class Node:
4 def __init__(self, val = 0, neighbors = None):
5 self.val = val
6 self.neighbors = neighbors if neighbors is not None else []
7"""
8
9import collections
10from typing import Optional
11
12class Solution:
13 def cloneGraph(self, node: Optional['Node']) -> Optional['Node']:
14 if node is None:
15 return None
16
17 queue = collections.deque([node])
18 maps = {}
19 while len(queue) > 0:
20 curr = queue.popleft()
21 maps[curr] = Node(curr.val)
22 for nei in curr.neighbors:
23 if nei not in maps:
24 queue.append(nei)
25
26 for old_node, new_node in maps.items():
27 for old_nei in old_node.neighbors:
28 newNei = maps[old_nei]
29 new_node.neighbors.append(newNei)
30 return maps[node]
