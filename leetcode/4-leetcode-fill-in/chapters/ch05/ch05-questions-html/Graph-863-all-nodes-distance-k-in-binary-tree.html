863.All Nodes Distance K in Binary Tree

1# Definition for a binary tree node.

8class Solution:
9 def distanceK(self, root: TreeNode, target: TreeNode, k: int) -> List[int]:
10 graph = collections.defaultdict(list)
11 self.dfs(root, graph, None)
12
13 queue = collections.deque([target])
14 visited = set()
15 visited.add(target)
16
17 while len(queue) > 0 and k >= 0:
18 size = len(queue)
19 res = []
20 for _ in range(size):
21 curr = queue.popleft()
22 res.append(curr.val)
23 for neighbour in graph[curr]:
24 if neighbour not in visited:
25 visited.add(neighbour)
26 queue.append(neighbour)
27 k -= 1
28
29 if k == -1:
30 return res
31 else:
32 return []

34 def dfs(self, node, graph, parent):
35 if parent:
36 graph[node].append(parent)
37 if node.left:
38 graph[node].append(node.left)
39 self.dfs(node.left, graph, node)
40 if node.right:
41 graph[node].append(node.right)
42 self.dfs(node.right, graph, node)
